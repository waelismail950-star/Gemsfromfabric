/** * GemsFromFabric — Apps Script backend. * * This file is version-controlled here for reference, but Apps Script * projects aren't deployed from git — paste this into the Apps Script * editor bound to your Google Sheet and deploy it as a Web App by hand. * See apps-script/README.md for the exact steps. * * The "Products" sheet IS the catalog. There is no hardcoded product * list anywhere else — add a row there (with a unique Product ID) and * it's a new item on the site on next page load; edit any cell (price, * size, name, stock, images, description) and that change is live the * same way. No code edit, no redeploy, no GitHub, for any of that. * * Responsibilities: * - doPost (default action): places an order — validates stock, * decrements it, appends a row to "Orders" (including which payment * method the customer picked, cod or whish), and sends the existing * email notification. * - doPost (action=updateStatus): admin-only, sets an order's status * to "accepted" or "rejected" (and restores stock on reject). * - doGet (action=getProducts): public — the full catalog, read * straight from the "Products" sheet. * - doGet (action=getOrders): admin-only — all orders as JSON. * * Required Script Property: * ADMIN_SECRET — whatever code you want to require as ?admin=CODE * on the site and as the "secret" param here. * Required Script Property (order emails silently fail without this): * SHOP_EMAIL — where order notifications go, e.g. your shop's Gmail. */ const ORDERS_SHEET_NAME = "Orders"; const PRODUCTS_SHEET_NAME = "Products"; const ORDERS_HEADER = ["Order ID", "Timestamp", "Items", "Total", "Phone", "Name", "Note", "Status", "Payment Method"]; const PRODUCTS_HEADER = ["Product ID", "Brand", "Name", "Category", "Size", "Price", "Stock", "Image URLs", "Description"]; // Column numbers (1-indexed, matches PRODUCTS_HEADER order) — used for // getRange() calls when updating a single cell (e.g. decrementing stock). const COL = { ID: 1, BRAND: 2, NAME: 3, CATEGORY: 4, SIZE: 5, PRICE: 6, STOCK: 7, IMAGES: 8, DESCRIPTION: 9 }; function getOrdersSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(ORDERS_SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(ORDERS_SHEET_NAME); sheet.appendRow(ORDERS_HEADER); } return sheet; } function getProductsSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(PRODUCTS_SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(PRODUCTS_SHEET_NAME); sheet.appendRow(PRODUCTS_HEADER); } return sheet; } function checkSecret_(secret) { const stored = PropertiesService.getScriptProperties().getProperty("ADMIN_SECRET"); return !!stored && !!secret && secret === stored; } function jsonOutput_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } function safeParseJson_(str) { try { return JSON.parse(str); } catch (e) { return []; } } /* ------------------------------------------------------------------ */ /* doGet — getProducts (public) and getOrders (admin-only) */ /* ------------------------------------------------------------------ */ function doGet(e) { const action = e.parameter.action; if (action === "getProducts") { return jsonOutput_({ products: readProductsList_() }); } if (action === "getOrders") { if (!checkSecret_(e.parameter.secret)) { return jsonOutput_({ error: "unauthorized" }); } return jsonOutput_({ orders: readOrders_() }); } return jsonOutput_({ error: "unknown action" }); } // Reads every row of "Products" into the shape the site expects. A row // with no Product ID is skipped (that's how you "delete" a listing // without deleting the row — clear column A). Numbers are coerced since // Sheets sometimes hands back cell values as strings. function readProductsList_() { const sheet = getProductsSheet_(); const values = sheet.getDataRange().getValues(); const out = []; for (let i = 1; i < values.length; i++) { const row = values[i]; const id = row[COL.ID - 1]; if (id === "" || id === null || id === undefined) continue; out.push({ id: Number(id), brand: row[COL.BRAND - 1] || "", name: row[COL.NAME - 1] || "", category: String(row[COL.CATEGORY - 1] || "").trim().toLowerCase(), size: row[COL.SIZE - 1] || "", price: Number(row[COL.PRICE - 1]) || 0, stock: Number(row[COL.STOCK - 1]) || 0, images: String(row[COL.IMAGES - 1] || "").split(",").map(s => s.trim()).filter(Boolean), description: row[COL.DESCRIPTION - 1] || "" }); } return out; } function readOrders_() { const sheet = getOrdersSheet_(); const values = sheet.getDataRange().getValues(); const rows = values.slice(1); return rows .filter(row => row[0] !== "" && row[0] !== null) .map(row => ({ id: row[0], timestamp: row[1], items: safeParseJson_(row[2]), total: row[3], phone: row[4], name: row[5], note: row[6], status: row[7], paymentMethod: row[8] || "cod" })) .reverse(); // newest first } /* ------------------------------------------------------------------ */ /* doPost — placeOrder (default) and updateStatus (admin-only) */ /* ------------------------------------------------------------------ */ function doPost(e) { const action = e.parameter.action || "placeOrder"; if (action === "updateStatus") { return handleUpdateStatus_(e); } return handlePlaceOrder_(e); } function buildStockMap_(productsSheet) { const values = productsSheet.getDataRange().getValues(); const map = {}; for (let i = 1; i < values.length; i++) { const id = values[i][COL.ID - 1]; if (id === "" || id === null) continue; map[id] = { row: i + 1, stock: Number(values[i][COL.STOCK - 1]) || 0 }; } return map; } function handlePlaceOrder_(e) { const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const items = safeParseJson_(e.parameter.items || "[]"); const productsSheet = getProductsSheet_(); const stockMap = buildStockMap_(productsSheet); // Check every item's stock BEFORE writing anything. const insufficient = []; items.forEach(item => { const entry = stockMap[item.id]; const available = entry ? entry.stock : 0; if (available < item.qty) { insufficient.push({ id: item.id, name: item.name, available: available, requested: item.qty }); } }); if (insufficient.length > 0) { return jsonOutput_({ status: "rejected", reason: "insufficient_stock", items: insufficient }); } // All good — decrement stock for real. items.forEach(item => { const entry = stockMap[item.id]; if (entry) { productsSheet.getRange(entry.row, COL.STOCK).setValue(entry.stock - item.qty); } }); const orderId = e.parameter.orderId || ("ORD-" + Date.now()); getOrdersSheet_().appendRow([ orderId, new Date().toISOString(), JSON.stringify(items), e.parameter.total || "", e.parameter.phone || "", e.parameter.name || "", e.parameter.note || "", "pending", e.parameter.paymentMethod || "cod" ]); // The order is already recorded and stock already decremented above — // a mail failure past this point shouldn't turn into a 500 that makes // the browser think the whole order failed (it already succeeded). let emailError = null; try { sendOrderEmail_(e.parameter.subject, e.parameter.orderText); } catch (err) { emailError = String(err); } return jsonOutput_({ status: "ok", orderId: orderId, emailError: emailError }); } finally { lock.releaseLock(); } } function sendOrderEmail_(subject, body) { // Session.getActiveUser() returns "" for anonymous web app requests (which // is what every real customer's checkout is), so that fallback silently // fails — always set the SHOP_EMAIL script property rather than relying on // it. getEffectiveUser() is kept as a second fallback since it at least // resolves to whoever the script is deployed to run as ("Execute as: Me"). const shopEmail = PropertiesService.getScriptProperties().getProperty("SHOP_EMAIL") || Session.getEffectiveUser().getEmail(); if (!shopEmail) { throw new Error("No SHOP_EMAIL script property set and couldn't resolve a fallback address."); } MailApp.sendEmail(shopEmail, subject || "New order", body || ""); } function handleUpdateStatus_(e) { if (!checkSecret_(e.parameter.secret)) { return jsonOutput_({ error: "unauthorized" }); } const orderId = e.parameter.orderId; const status = e.parameter.status; if (!orderId || (status !== "accepted" && status !== "rejected")) { return jsonOutput_({ error: "bad request" }); } const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const sheet = getOrdersSheet_(); const values = sheet.getDataRange().getValues(); for (let i = 1; i < values.length; i++) { if (values[i][0] === orderId) { sheet.getRange(i + 1, 8).setValue(status); // column 8 = Status if (status === "rejected") { restoreStockForOrder_(safeParseJson_(values[i][2])); } return jsonOutput_({ status: "ok" }); } } return jsonOutput_({ error: "order not found" }); } finally { lock.releaseLock(); } } function restoreStockForOrder_(items) { const productsSheet = getProductsSheet_(); const stockMap = buildStockMap_(productsSheet); items.forEach(item => { const entry = stockMap[item.id]; if (entry) { productsSheet.getRange(entry.row, COL.STOCK).setValue(entry.stock + item.qty); } }); }
